Agregación
CREATE TABLE aggregations
STORED AS parquet
LOCATION '/data/aggregations'
AS SELECT COUNT(rutas.destinationairport) AS total_destinations, aeropuertos.country AS country  
FROM rutas
JOIN aerolineas ON rutas.airlineid = aerolineas.airlineid
JOIN aeropuertos ON aeropuertos.airportid = rutas.destinationid
WHERE rutas.stops = 0 AND aeropuertos.altitude > 5000 AND aerolineas.active = False
GROUP BY aeropuertos.country;
